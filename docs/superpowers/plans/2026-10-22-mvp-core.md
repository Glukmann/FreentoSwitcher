# FreentoSwitcher — Plan A: рабочее ядро (MVP)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Запускаемое меню-барное приложение для macOS arm64: переключение RU↔EN по хоткею, автоконвертация слова по разделителю, звуки, автозапуск.

**Architecture:** SwiftPM-пакет из трёх таргетов: `Engine` (чистая логика, без AppKit), `Platform` (CGEventTap, TIS, синтез клавиш), `App` (исполняемый таргет: статус-бар, меню, wiring). Логика — state machine в Engine, выдающая эффекты; Platform/App интерпретируют.

**Tech Stack:** Swift 6 (swift-tools 6.0, strict concurrency по умолчанию), swift-testing (`import Testing`), AppKit/CoreGraphics/TextInputSources (TIS через CoreServices), SMAppService. Без внешних зависимостей.

**Spec:** `docs/design/2026-10-22-architecture.md` (разделы 4–8, функции A1–A3, A9 (одноуровневый undo), A13, A15, A17, A18; A4 (частичная автоконвертация), A7–A8, A10-флаги частично, A14, A16 — в Plan B).

```swift
public struct KeyStrokeBuffer: Equatable, Sendable {
    public static let maxLength = 32

    /// Punctuation keycodes that act as word separators when counting.
    public static let punctuationKeycodes: Set<UInt16> = [0x2C, 0x2E, 0x2F]

    public private(set) var strokes: [KeyStroke] = []
    public var layoutID: String = ""

    public var count: Int { strokes.count }

    public var punctuationMarksCount: Int {
        strokes.filter { Self.punctuationKeycodes.contains($0.keycode) }.count
    }

    public var beginsWithPunctuation: Bool {
        strokes.first.map { Self.punctuationKeycodes.contains($0.keycode) } ?? false
    }

    public init() {}

    public mutating func append(_ stroke: KeyStroke) {
        strokes.append(stroke)
        if strokes.count > Self.maxLength {
            strokes.removeFirst(strokes.count - Self.maxLength)
        }
    }

    public mutating func removeLast() {
        if !strokes.isEmpty { strokes.removeLast() }
    }

    public mutating func removeAll() {
        strokes.removeAll()
    }

    public func stringValue(convertingWith engine: ConversionEngine) -> String {
        engine.string(for: strokes, layoutID: layoutID)
    }
}
```

`Sources/Engine/Conversion/ConversionEngine.swift`:
```swift
public enum ConversionMode: Equatable, Sendable {
    case toRussian
    case toEnglish
}

public struct CharacterPairMap: Equatable, Sendable {
    private let pairs: [Character: Character]

    public init(pairs: [Character: Character]) {
        self.pairs = pairs
    }

    public func convert(_ c: Character) -> Character {
        pairs[c] ?? c
    }
}

public struct ConversionEngine: Sendable {
    private let russianToEnglish: CharacterPairMap
    private let englishToRussian: CharacterPairMap
    private let keycodeToCharRU: [UInt16: Character]
    private let keycodeToCharEN: [UInt16: Character]

    public init(russianToEnglish: CharacterPairMap,
                englishToRussian: CharacterPairMap,
                keycodeToCharRU: [UInt16: Character],
                keycodeToCharEN: [UInt16: Character]) {
        self.russianToEnglish = russianToEnglish
        self.englishToRussian = englishToRussian
        self.keycodeToCharRU = keycodeToCharRU
        self.keycodeToCharEN = keycodeToCharEN
    }

    public func convert(_ string: String, mode: ConversionMode) -> String {
        let map = mode == .toEnglish ? russianToEnglish : englishToRussian
        return String(string.map { map.convert($0) })
    }

    public func string(for strokes: [KeyStroke], layoutID: String) -> String {
        let table = isRussianLayoutID(layoutID) ? keycodeToCharRU : keycodeToCharEN
        return String(strokes.compactMap { table[$0.keycode] })
    }

    private func isRussianLayoutID(_ id: String) -> Bool {
        id.localizedCaseInsensitiveContains("russian") || id.localizedCaseInsensitiveContains("рус")
    }
}
```

```swift
import Foundation

public enum LayoutVerdict: Equatable, Sendable {
    case unknown
    case incorrectLayout
    case exception
}

public struct LayoutChecker: Sendable {
    private let triggers: Set<String>
    private let userRules: [UserRule]

    public init(triggers: [String], userRules: [UserRule]) {
        self.triggers = Set(triggers.map { $0.lowercased() })
        self.userRules = userRules.filter(\.isActive)
    }

    public func check(_ word: String, isEnglishLayout: Bool) -> LayoutVerdict {
        let w = word.lowercased()
        // 1. user exceptions (block conversion)
        for rule in userRules where !rule.shouldSwitch {
            if matches(rule, word: w) { return .exception }
        }
        // 2. user force-switch dictionary
        for rule in userRules where rule.shouldSwitch {
            if matches(rule, word: w) { return .incorrectLayout }
        }
        // 3. trigger words
        if triggers.contains(w) { return .incorrectLayout }
        return .unknown
    }

    private func matches(_ rule: UserRule, word: String) -> Bool {
        if rule.isRegex {
            return word.range(of: rule.pattern, options: .regularExpression) != nil
        }
        return word == rule.pattern.lowercased()
    }
}
```

```swift
public struct WordBoundaryDetector: Sendable {
    /// Keycodes that finish a word and trigger the autocorrection check.
    /// 0x31 space, 0x24 enter, 0x30 tab, 0x2B comma, 0x2F period,
    /// 0x2C slash, 0x27 hyphen, 0x2D n, 0x29 semicolon, 0x28 quote.

        0x31, 0x24, 0x30, 0x2B, 0x2F, 0x2C, 0x27, 0x2D, 0x29, 0x28,
    ]

        self.noAutoconvertAfterTabOrEnter = noAutoconvertAfterTabOrEnter
    }

    public func isWordSeparator(_ keycode: UInt16) -> Bool {
        if noAutoconvertAfterTabOrEnter && (keycode == 0x30 || keycode == 0x24) {
            return false
        }
        return Self.separatorKeycodes.contains(keycode)
```

`Sources/Engine/StateMachine/TypingStateMachine.swift`:
```swift
public struct TypingStateMachine: Sendable {
    public private(set) var rawBuffer = KeyStrokeBuffer()
    private var undoBuffer = KeyStrokeBuffer()
    private var settings: AppSettings
    private let engine: ConversionEngine
    private let checker: LayoutChecker
    private var cancelNextWord = false
    public private(set) var autocorrectionIsActive: Bool

    public init(settings: AppSettings, engine: ConversionEngine, checker: LayoutChecker) {
        self.settings = settings
        self.engine = engine
        self.checker = checker
        self.autocorrectionIsActive = settings.isAutocorrectionActive
    }

    public mutating func handleKeyDown(_ key: KeyStroke, isSeparator: Bool) -> [EngineEffect] {
        if key.keycode == 0x35 { return reset() }                    // Escape
        if key.keycode == 0x33 {                                     // Backspace
            rawBuffer.removeLast()
            undoBuffer.removeLast()
            return [.none]
        }
        guard autocorrectionIsActive else { return [.none] }
        if isSeparator {
            return tryAutoconvert()
        }
        rawBuffer.append(key)
        undoBuffer.append(key)
        return [.none]
    }

    public mutating func handleManualLayoutSwitch() -> [EngineEffect] {
        if settings.cancelNextWordAfterManualSwitch { cancelNextWord = true }
        return [.switchLayout, .playSound(.switchToEnglish)] // направление звука уточняет App
    }

    public mutating func handleEscape() -> [EngineEffect] { reset() }

    public mutating func handleBackspace() -> [EngineEffect] {
        rawBuffer.removeLast()
        return [.none]
    }

    public mutating func handleUndoHotkey() -> [EngineEffect] {
        guard undoBuffer.count > 0 else { return [.none] }
        let effects: [EngineEffect] = [
            .switchLayout, .deleteCharacters(count: undoBuffer.count),
            .replayBuffer, .playSound(.undo)]
        undoBuffer.removeAll()
        return effects
    }

    public mutating func toggleAutocorrection() -> [EngineEffect] {
        autocorrectionIsActive.toggle()
        return [.none]
    }

    private mutating func tryAutoconvert() -> [EngineEffect] {
        defer { rawBuffer.removeAll() }
        guard rawBuffer.count > 0 else { return [.none] }
        if cancelNextWord {
            cancelNextWord = false
            return [.resetBuffers]
        }
        let word = rawBuffer.stringValue(convertingWith: engine)
        let verdict = checker.check(word, isEnglishLayout: true)
        switch verdict {
        case .incorrectLayout:
            return [.switchLayout, .deleteCharacters(count: rawBuffer.count),
                    .replayBuffer, .playSound(.autoConvert), .resetBuffers]
        case .unknown, .exception:
            return [.resetBuffers]
        }
    }

    private mutating func reset() -> [EngineEffect] {
        rawBuffer.removeAll()
        undoBuffer.removeAll()
        return [.resetBuffers]
    }
}
```
Примечание: `.resetBuffers` в конце списка эффектов конвертации — App сбрасывает UI-состояние; буферы уже сброшены в `defer`.

```swift
private func charsForLayout(_ layoutID: String) -> [UInt16: Character] {
    guard let source = source(forLayoutID: layoutID),
          let data = source.value(forProperty: kTISPropertyUnicodeKeyboardLayoutData) as? Data
    else { return [:] }

    let kbdType = UInt32(LMGetKbdType())

            var length = 0

                                        kbdType, 0, &deadKeyState,
                                        chars.count, &length, &chars)
            if status == noErr, length == 1, let scalar = Unicode.Scalar(chars[0]) {
                result[keycode] = Character(scalar)
            }
        }
    }
    return result
```

```swift
            eventsOfInterest: CGEventMask((1 << CGEventType.keyDown.rawValue)
                | (1 << CGEventType.keyUp.rawValue)
                | (1 << CGEventType.flagsChanged.rawValue)),
            callback: callback,
            userInfo: Unmanaged.passUnretained(self).toOpaque()
        ) else { return false }
        machPort = port
        runLoopSource = CFMachPortCreateRunLoopSource(kCFAllocatorDefault, port, 0)
        CFRunLoopAddSource(CFRunLoopGetCurrent(), runLoopSource, .commonModes)
        CGEvent.tapEnable(tap: port, enable: true)
        return true
    }

    public func stop() {
        if let port = machPort { CGEvent.tapEnable(tap: port, enable: false) }
        if let src = runLoopSource { CFRunLoopRemoveSource(CFRunLoopGetCurrent(), src, .commonModes) }
        machPort = nil; runLoopSource = nil
    }

    public var isSecureInputWarningNeeded: Bool {
        IsSecureEventInputEnabled()
    }

    private func handleEvent(type: CGEventType, event: CGEvent) -> Unmanaged<CGEvent>? {
        if type == .tapDisabledByTimeout || type == .tapDisabledByUserInput {
            if let port = machPort { CGEvent.tapEnable(tap: port, enable: true) }
            return nil
        }
        if event.getIntegerValueField(.eventSourceUserData) == Self.ownEventUserData {
            return Unmanaged.passRetained(event) // наши собственные события — не обрабатываем, но пропускаем
        }
        guard type == .keyDown || type == .keyUp else {
            return Unmanaged.passRetained(event)
        }
        let key = KeyEvent(
            keycode: UInt16(event.getIntegerValueField(.keyboardEventKeycode)),
            modifiers: event.flags.rawValue,
            type: type == .keyDown ? .keyDown : .keyUp)
        let passThrough = delegate?.eventTap(self, didReceiveKey: key) ?? true
        return passThrough ? Unmanaged.passRetained(event) : nil
    }
}
```
Для поглощения событий (нужно в Plan B для конвертации выделения) listenOnly-тап не может возвращать nil; в Task 10 оставляем listenOnly (MVP хватает: мы не подавляем нажатия пользователя, только дополняем). import AppKit
```swift
import Engine
import Platform

final class AppController: NSObject, EventTapDelegate {
    private let settings: AppSettings
    private let inputSources = InputSourceManager.shared
    private let eventTap = EventTap()
    private let sounds: SoundPlayer
    private var stateMachine: TypingStateMachine
    private let boundaries: WordBoundaryDetector
    private let statusBar = StatusBarController()

    override init() {
        settings = SettingsStore.load()
        let maps = InputSourceManager.shared.makeKeycodeMaps(
            russianLayoutID: settings.russianLayoutID,
            englishLayoutID: settings.englishLayoutID)
        let engine = ConversionEngine(
            russianToEnglish: CharacterPairMap(pairs: buildPairs(maps.keycodeToCharRU, maps.keycodeToCharEN)),
            englishToRussian: CharacterPairMap(pairs: buildPairs(maps.keycodeToCharEN, maps.keycodeToCharRU)),
            keycodeToCharRU: maps.keycodeToCharRU,
            keycodeToCharEN: maps.keycodeToCharEN)
        stateMachine = TypingStateMachine(
            settings: settings,
            engine: engine,
            checker: LayoutChecker(triggers: loadSeedTriggers(), userRules: settings.userRules))
        boundaries = WordBoundaryDetector(noAutoconvertAfterTabOrEnter: settings.noAutoconvertAfterTabOrEnter)
        sounds = SoundPlayer(settings: settings)
        super.init()
        eventTap.delegate = self
        statusBar.onSwitchLayout = { [weak self] in self?.manualSwitch() }
        statusBar.onToggleAutocorrection = { [weak self] in
            _ = self?.stateMachine.toggleAutocorrection()
        }
        statusBar.onQuit = { NSApplication.shared.terminate(nil) }
    }

    func start() {
        guard AXIsProcessTrusted() else {
            // Plan B: полноценный онбординг; MVP: предупреждение в лог
            NSLog("FreentoSwitcher: Accessibility permission not granted")
        }
        _ = eventTap.start()
    }

    func eventTap(_ tap: EventTap, didReceiveKey key: KeyEvent) -> Bool {
        guard key.type == .keyDown else { return true }
        // A15: исключённые приложения — сравнение bundleID фронт-процесса (MVP через NSWorkspace)
        let frontBundle = NSWorkspace.shared.frontmostApplication?.bundleIdentifier ?? ""
        if settings.excludedApps.contains(frontBundle) { return true }
        // A2: хоткей ручного переключения
        if key.keycode == settings.hotkeySwitchLayout.keycode,
           key.modifiers & 0x1C0000 == settings.hotkeySwitchLayout.modifiers & 0x1C0000 {
            apply(stateMachine.handleManualLayoutSwitch())
            return true
        }
        // A9: хоткей отмены конвертации
        if key.keycode == settings.hotkeyUndo.keycode,
           settings.hotkeyUndo.keycode != 0xFFFF,
           key.modifiers & 0x1C0000 == settings.hotkeyUndo.modifiers & 0x1C0000 {
            apply(stateMachine.handleUndoHotkey())
            return true
        }
        let isSep = boundaries.isWordSeparator(key.keycode)
        apply(stateMachine.handleKeyDown(
            KeyStroke(keycode: key.keycode, modifiers: key.modifiers), isSeparator: isSep))
        return true
    }

    private func manualSwitch() {
        apply(stateMachine.handleManualLayoutSwitch())
    }

    private func apply(_ effects: [EngineEffect]) {
        for e in effects {
            switch e {
            case .switchLayout:
                inputSources.switchLanguage(
                    russianLayoutID: settings.russianLayoutID,
                    englishLayoutID: settings.englishLayoutID)
            case .playSound(let mask):
                sounds.play(mask, switchingToRussian: inputSources.isCurrentRussian())
            case .deleteCharacters(let count):
                for _ in 0..<count { EventSynth.pressKey(0x33) }
            case .replayBuffer:
                for s in stateMachine.rawBuffer.strokes { EventSynth.pressKey(s.keycode) }
            case .resetBuffers: break
            case .toggleAutocorrection: break
            case .none: break
            }
        }
    }

    private func buildPairs(_ from: [UInt16: Character], _ to: [UInt16: Character]) -> [Character: Character] {
        var pairs: [Character: Character] = [:]
        for (kc, dst) in to {
            if let src = from[kc], src != dst { pairs[src] = dst }
        }
        return pairs
    }

    private func loadSeedTriggers() -> [String] {
        guard let url = Bundle.module.url(forResource: "seed-triggers", withExtension: "txt"),
              let s = try? String(contentsOf: url) else { return [] }
        return s.split(separator: "\n").map(String.init)
    }
}
```
